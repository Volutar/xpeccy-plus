# Keyboard bench cases: a host key that is a Spectrum combination (. = SS+M, Delete =
# CS+0) has its shift down before the key and up after it; ours go whole. Typed (Shift+;
# as :): the host's Shift is hidden while the symbol is down, in any order of the two.
#
#   bin  <file in bin/>  <source in asm/>         how mkbins.py assembles it
#   run  <case>  <file in bin/>  <presses>  [ENV=value ...]

bin  poll.bin  poll.asm

run  48k-host   poll.bin  20  MACHINE=ZX48 KEYS=host
run  48k-ours   poll.bin  20  MACHINE=ZX48 KEYS=ours
run  pent-host  poll.bin  20  MACHINE=Pentagon KEYS=host
run  next-host  poll.bin  20  MACHINE=ZXNext KEYS=host
run  48k-typed-first    poll.bin  10  MACHINE=ZX48 KEYS=typed ORDER=first
run  48k-typed-late     poll.bin  10  MACHINE=ZX48 KEYS=typed ORDER=late
run  48k-typed-upfirst  poll.bin  10  MACHINE=ZX48 KEYS=typed ORDER=upfirst
run  48k-typed-held     poll.bin  10  MACHINE=ZX48 KEYS=typed ORDER=held
run  pent-typed-held    poll.bin  10  MACHINE=Pentagon KEYS=typed ORDER=held
run  profi-typed-held   poll.bin  10  MACHINE=Profi KEYS=typed ORDER=held
run  evo-typed-held     poll.bin  10  MACHINE=Baseconf KEYS=typed ORDER=held
run  evo-typed-late     poll.bin  10  MACHINE=Baseconf KEYS=typed ORDER=late
run  48k-pos-first      poll.bin  10  MACHINE=ZX48 KEYS=typed ORDER=first CHAR=0
run  48k-typed-plain    poll.bin  10  MACHINE=ZX48 KEYS=typed ORDER=plain
run  next-ps2-equal    poll.bin  1   MACHINE=ZXNext KEYS=ps2 ATCODE=0x55
run  next-ps2-o        poll.bin  1   MACHINE=ZXNext KEYS=ps2 ATCODE=0x44
# The Kempston mouse's ports: decoded on A5, A8, A9 and A10, as the interface's schematic;
# the left button on D1 there, on D0 on the Evo (its AVR)
run  48k-mouse          poll.bin  0   MACHINE=ZX48 KEYS=mouse
run  128k-mouse         poll.bin  0   MACHINE=ZX128 KEYS=mouse
run  2a-mouse           poll.bin  0   MACHINE=Plus2A KEYS=mouse
run  p3-mouse           poll.bin  0   MACHINE=Plus3 KEYS=mouse
run  p3-mouse-off       poll.bin  0   MACHINE=Plus3 KEYS=mouse MOUSE=0
run  2a-mouse-off       poll.bin  0   MACHINE=Plus2A KEYS=mouse MOUSE=0
run  128k-mouse-off     poll.bin  0   MACHINE=ZX128 KEYS=mouse MOUSE=0
run  p3-mouse-swap      poll.bin  0   MACHINE=Plus3 KEYS=mouse SWAP=1
run  evo-mouse          poll.bin  0   MACHINE=Baseconf KEYS=mouse
run  ts-mouse           poll.bin  0   MACHINE=TSConf KEYS=mouse
run  ts-mouse-swap      poll.bin  0   MACHINE=TSConf KEYS=mouse SWAP=1
