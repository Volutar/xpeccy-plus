# chapter 5: CIPSEND / CIPSENDEX: forms, diagnostics, busy s..., \0 and \\, CIPDINFO's +IPD,
# CLOSED after the data, UDP (local port, a target, its diagnostics)
mask .*\\x[0-9A-F][0-9A-F].* <romlog>
mask ,\d{5},0\\r\\n$ ,<lport>,0\r\n
wait WIFI GOT IP\r\n 5000
send ATE0
wait \r\nOK\r\n
send AT+CIPSEND=5
wait \r\nERROR\r\n
send AT+CIPSEND=?
wait \r\nOK\r\n
send AT+CIPSEND?
wait \r\nERROR\r\n
send AT+CIPSENDEX=?
wait \r\nOK\r\n
send AT+CIPSENDEX
wait \r\nERROR\r\n
server echo
send AT+CIPSTART="TCP","127.0.0.1",<port>
wait \r\nOK\r\n
send AT+CIPSEND=2049
wait \r\nERROR\r\n
send AT+CIPSEND=5,1
wait \r\nERROR\r\n
send AT+CIPSEND=abc
wait \r\nERROR\r\n
send AT+CIPSEND=5,"127.0.0.1",80
wait \r\nERROR\r\n
send AT+CIPSEND=0
wait \r\nERROR\r\n
send AT+CIPSEND5
wait \r\nERROR\r\n
# three bytes asked, five sent: the last two come while it waits for SEND OK
send AT+CIPSEND=3
wait >\x20
raw abc\r\n
wait SEND OK\r\n
quiet 300
send AT+CIPSENDEX=10
wait >\x20
raw ab\\0
wait SEND OK\r\n
quiet 300
send AT+CIPSENDEX=3
wait >\x20
raw a\\\\b
wait SEND OK\r\n
quiet 300
send AT+CIPDINFO=1
wait \r\nOK\r\n
send AT+CIPSEND=2
wait >\x20
raw hi
wait SEND OK\r\n
quiet 300
send AT+CIPDINFO=0
wait \r\nOK\r\n
send AT+CIPCLOSE
wait \r\nOK\r\n
# the line server answers and closes: the data, then CLOSED
server line

send AT+CIPSEND=7
wait >\x20
raw hello\r\n
wait CLOSED\r\n
send AT+CIPSEND=1
wait \r\nERROR\r\n
