# A test chain for the shader engine: every kind of thing a pass can ask for, each in a
# form a window shot can prove. Not a look. See README.md beside it.
#
#   bright, back   values above 1 kept by a float target and brought back: the picture
#                  comes out as it went in. Without a float target it comes out dark
#                  (everything above a quarter is cut to a quarter).
#   ghost          the frame three back under the one shown: a moving thing is there twice.
#   trail          its own last output, faded: a moving thing leaves a tail.
#   grid           under "when GRIDON": a texture laid over the picture, or no pass at all.
#   the last pass  the machine's numbers as rows of 16 squares at the picture's bottom left.

#pragma parameter TRAIL "Trail length" 0.70 0.0 0.99 0.01
#pragma parameter GRIDON "Grid pass" 1.0 0.0 1.0 1.0
texture GRID = grid.png nearest repeat

common:
	vec3 src(vec2 uv) { return texture(Source, uv).rgb; }

pass bright:
	format rgba16f
fragment:
	in vec2 vTexCoord;
	layout(location = 0) out vec4 FragColor;
	void main() {
		FragColor = vec4(src(vTexCoord) * 4.0, 1.0);
	}

pass back:
fragment:
	in vec2 vTexCoord;
	layout(location = 0) out vec4 FragColor;
	void main() {
		FragColor = vec4(src(vTexCoord) / 4.0, 1.0);
	}

pass ghost:
fragment:
	in vec2 vTexCoord;
	layout(location = 0) out vec4 FragColor;
	void main() {
		vec3 was = texture(OriginalHistory3, vTexCoord).rgb;
		FragColor = vec4(max(src(vTexCoord), was * 0.6), 1.0);
	}

pass trail:
fragment:
	in vec2 vTexCoord;
	layout(location = 0) out vec4 FragColor;
	void main() {
		vec3 old = texture(trailFeedback, vTexCoord).rgb;
		FragColor = vec4(max(src(vTexCoord), old * TRAIL), 1.0);
	}

pass grid:
	scale viewport 1.0
	when GRIDON
fragment:
	in vec2 vTexCoord;
	layout(location = 0) out vec4 FragColor;
	void main() {
		vec3 m = texture(GRID, gl_FragCoord.xy * GRIDSize.zw).rgb;
		FragColor = vec4(src(vTexCoord) * m, 1.0);
	}

pass:
fragment:
	in vec2 vTexCoord;
	layout(location = 0) out vec4 FragColor;

	// row by row from the bottom, 16 bits each, the high bit at the left
	uint row(int n) {
		if (n == 0) return uint(xpRaster.x);			// dots a line
		if (n == 1) return uint(xpRaster.y);			// lines a frame
		if (n == 2) return uint(xpRaster.z);			// the first shown dot
		if (n == 3) return uint(xpRaster.w);			// the first shown line
		if (n == 4) return uint(xpCarrier.x * 65536.0 + 0.5);	// carrier, cycles a dot
		if (n == 5) return uint(xpCarrier.y) * 2u + uint(xpCarrier.w);	// locked, V sense
		if (n == 6) return uint(xpCarrier.z * 65535.0 + 0.5);	// carrier's phase
		if (n == 7) return uint(xpFrame) & 65535u;		// the machine's frame
		if (n == 8) return FrameCount & 65535u;			// paints
		if (n == 9) return uint(OriginalSize.x);		// the frame's texels across
		if (n == 10) return uint(FinalViewportSize.x);		// the picture's pixels across
		return 0u;
	}

	void main() {
		vec3 c = src(vTexCoord);
		vec2 p = gl_FragCoord.xy - xpPicture.xy;
		ivec2 cell = ivec2(floor(p / 8.0));
		if ((p.x >= 0.0) && (p.y >= 0.0) && (cell.x < 16) && (cell.y < 11)) {
			vec2 f = p - vec2(cell) * 8.0;
			bool on = ((row(cell.y) >> uint(15 - cell.x)) & 1u) != 0u;
			if ((f.x >= 1.0) && (f.y >= 1.0)) c = on ? vec3(1.0, 1.0, 0.0) : vec3(0.0, 0.0, 0.5);
		}
		FragColor = vec4(c, 1.0);
	}
