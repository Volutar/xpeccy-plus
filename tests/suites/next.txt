# ZX Spectrum Next on System/Next 24.11 trimmed to what NextZXOS boots from (corpus
# next/sysnext-24.11-min: machines/next's ROMs, menu.def and keymap.bin, nextzxos/, sys/).
# NextZXOS's menu is up by frame 700; the board's keys go in there (pc: = its PS/2 keyboard).
#
# name          machine  how   skip  frames  file  options
next_boot       zxnext   boot  850   50            sd:next/sysnext-24.11-min
# the NMI button from the menu, and F9 let go at once: NextZXOS's NMI menu
next_nmi        zxnext   boot  850   50            sd:next/sysnext-24.11-min keys:700:nmi
next_f9         zxnext   boot  850   50            sd:next/sysnext-24.11-min keys:700:pc:F9
# F9 held over a second: no NMI, the same hashes as next_boot
next_f9_long    zxnext   boot  850   50            sd:next/sysnext-24.11-min keys:700:pc:F9/70
# F10, DRIVE: NextZXOS turns it off (NR 06 b4), the same hashes as next_boot
next_f10        zxnext   boot  850   50            sd:next/sysnext-24.11-min keys:700:pc:F10
next_f3_60hz    zxnext   boot  850   50            sd:next/sysnext-24.11-min keys:700:pc:F3
next_f8_speed   zxnext   boot  850   50            sd:next/sysnext-24.11-min keys:700:pc:F8
next_f4_reset   zxnext   boot  850   50            sd:next/sysnext-24.11-min keys:700:pc:F4
# a digit held with F9 is a function key: F9+4 a soft reset, F9+1 a hard one (the firmware again)
next_f9_4       zxnext   boot  850   50            sd:next/sysnext-24.11-min keys:700:pc:F9/10,702:pc:4
next_f9_1       zxnext   boot  1150  50            sd:next/sysnext-24.11-min keys:700:pc:F9/10,702:pc:1
# 1 MB: an issue 2 board without its expansion - NextZXOS shows 768K, and MAXBNK (23401) is 47; 111 on 2 MB
next_boot_1m    zxnext   boot  850   50            sd:next/sysnext-24.11-min mac:machine.memory=1024
next_maxbnk_1m  zxnext   boot  1050  50            sd:next/sysnext-24.11-min mac:machine.memory=1024 keys:700:n,760:p,772:r,784:i,796:n,808:t,820:_,832:p,844:e,856:e,868:k,880:_,892:2,904:3,916:4,928:0,940:1,952:E
next_maxbnk_2m  zxnext   boot  1050  50            sd:next/sysnext-24.11-min keys:700:n,760:p,772:r,784:i,796:n,808:t,820:_,832:p,844:e,856:e,868:k,880:_,892:2,904:3,916:4,928:0,940:1,952:E
# a .nex holding the DMA by CTC 0 in hardware IM 2, its ISR's length sweeping across the CTC
# period: RETIs right where the channel fires again (the DMA's held answer, perf-weak-host).
# A Next program starts on a card of its own, System/Next (sys:) with its folder
next_reti_dma   zxnext   open  1500  100   next/dma/reti-ctc-dma.nex sys:next/sysnext-24.11-min put:dot/CD=next/dot/CD put:dot/NEXLOAD=next/dot/NEXLOAD
# ...whatever card is in the slot: an image here, the same hashes as next_reti_dma
next_reti_dma_img zxnext open 1500  100   next/dma/reti-ctc-dma.nex sdimg:next/sysnext-24.11-min sys:next/sysnext-24.11-min put:dot/CD=next/dot/CD put:dot/NEXLOAD=next/dot/NEXLOAD
# the Browser's place survives a reset, not a power cycle: _f4, _f9_1, _reset bless only with next_browser's frames
next_browser       zxnext   boot  2300  50            sd:next/sysnext-24.11-min keys:760:pc:ENT,900:pc:DOWN,930:pc:DOWN,960:pc:ENT,1010:pc:DOWN,1030:pc:DOWN,1050:pc:DOWN
next_browser_f4    zxnext   boot  2300  50            sd:next/sysnext-24.11-min keys:760:pc:ENT,900:pc:DOWN,930:pc:DOWN,960:pc:ENT,1010:pc:DOWN,1030:pc:DOWN,1050:pc:DOWN,1150:pc:F4,2150:pc:ENT
next_browser_f9_1  zxnext   boot  2300  50            sd:next/sysnext-24.11-min keys:760:pc:ENT,900:pc:DOWN,930:pc:DOWN,960:pc:ENT,1010:pc:DOWN,1030:pc:DOWN,1050:pc:DOWN,1150:pc:F9/10,1152:pc:1,2150:pc:ENT
next_browser_reset zxnext   boot  2300  50            sd:next/sysnext-24.11-min keys:760:pc:ENT,900:pc:DOWN,930:pc:DOWN,960:pc:ENT,1010:pc:DOWN,1030:pc:DOWN,1050:pc:DOWN,1150:reset,2150:pc:ENT
next_browser_power zxnext   boot  2300  50            sd:next/sysnext-24.11-min keys:760:pc:ENT,900:pc:DOWN,930:pc:DOWN,960:pc:ENT,1010:pc:DOWN,1030:pc:DOWN,1050:pc:DOWN,1150:power,2150:pc:ENT
