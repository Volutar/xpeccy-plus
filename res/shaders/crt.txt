# CRT - Xpeccy+
#
# A tube in stages, each drawn once at the size it needs; a stage whose knob is at 0 is
# not drawn at all:
#   afterglow  at the source's size: each color keeps the brighter of this frame and its
#              own last picture faded - the phosphor's decay
#   line       along the line, at the picture's width and the source's lines: how far a
#              dot spreads (the beam's focus), averaged over the screen pixel; the red and
#              blue beams parted from the green one; a faint second picture beside the
#              first (a ghost), lines pushed aside by interference
#   glowh, glowv  a quarter of the picture, blurred across and down: the light around
#              bright things
#   last       down the screen: the beam's profile (scanlines, flat where a line would
#              pass full light), the mask through which it lands, the screen's curve, its
#              corners and vignette
# Light is linear inside and stored gamma-encoded between the passes, so an 8-bit target
# holds it as well as a float one. What the scanlines and the mask take is given back.
#
# The two-pass beam follows Hyllian's CRT (MIT), the mask layouts Timothy Lottes's and
# hunterk's (public domain). The beams' parting towards the screen's edges is Volutar's.
# The ghost, the interference and the noise are after Mattias Gustavsson's newpixie CRT
# (MIT / public domain). Same MIT license as the rest of Xpeccy+.

common:
	// a mix's share snapped at its ends: in linear light a thousandth of white over
	// black is seen
	float snap(float k) { return clamp((k - 0.001) / 0.998, 0.0, 1.0); }

	// How far a beam lands from the green one at a point of the picture (0 .. 1 both
	// ways): dots across and lines down, growing from the screen's middle to its edges.
	// most: what it is at the edge
	vec2 parting(vec2 uv, vec2 most) {
		vec2 d = uv * 2.0 - 1.0;
		return DIVERGE * most * (DIVMID + (1.0 - DIVMID) * length(d)) * d;
	}

	// What moves with time: a turn (0 .. 2 pi) that goes round k times in 1000 frames
	// and so meets itself when the count of frames starts again
	float turn(uint frames, uint k) {
		return 6.28318530718 * float(((frames % 1000u) * k) % 1000u) / 1000.0;
	}

// The picture as a PAL set gets it over a composite or an aerial cable: each line made
// into luma with the color on a carrier, and taken apart again as a set does it. The
// encoder is the look's own choice, the same for every machine (a clone had none of its
// own, and what stood between it and the set was anybody's): a 128K's - 5/8 cycle a dot
// from the dots' own crystal, whole cycles a line, so the pattern stands; 311 lines, so
// the V switch, which turns every line, comes to a line the other way round each frame -
// or a 48K's - a crystal of its own, 283.7516 cycles a line, half a cycle on each frame,
// so the pattern crawls; 312 lines, the V switch the same each frame. Decoding: color by the carrier, low-passed and
// averaged with the line above (the delay line), luma less a band around the carrier.
// Fine black and white detail comes out colored and color edges crawl, as they did.
// Two texels a dot.
//
// First the encoder's own part: luma as it is, the two color differences low-passed -
// an encoder has no more than about 1.3 MHz of color to put on its carrier (a 128K's
// changes its color 5 times in 8 dots). Without this a pattern of two colors dot by dot
// would put its color on the carrier at the dots' rate and come back as stripes.
// Kept as luma and the two halved, about 128 of 255.
pass coder:
	when INPUT
	filter nearest
fragment:
	in vec2 vTexCoord;
	layout(location = 0) out vec4 FragColor;

	void main() {
		ivec2 at = ivec2(vTexCoord * SourceSize.xy);
		int last = int(SourceSize.x) - 1;
		vec3 here = texelFetch(Source, at, 0).rgb;
		float y = dot(here, vec3(0.299, 0.587, 0.114));
		vec2 uv = vec2(0.0);
		float all = 0.0;
		for (int k = -4; k <= 4; k++) {
			vec3 c = texelFetch(Source, ivec2(clamp(at.x + k, 0, last), at.y), 0).rgb;
			float yk = dot(c, vec3(0.299, 0.587, 0.114));
			// a bell 1.8 texels wide, brought down to nothing at the taps' ends
			float w = exp(-0.5 * float(k * k) / 3.24) * (0.5 + 0.5 * cos(0.62831853 * float(k)));
			uv += w * vec2(0.492 * (c.b - yk), 0.877 * (c.r - yk));
			all += w;
		}
		// (128 of 255 is a level the target has; one half is not, and would come back as a tint)
		FragColor = vec4(y, 128.0 / 255.0 + 0.5 * uv / all, 1.0);
	}

pass signal:
	when INPUT
	filter nearest
fragment:
	in vec2 vTexCoord;
	layout(location = 0) out vec4 FragColor;

	const float PI2 = 6.28318530718;
	const int TAPS = 10;

	// the carrier, cycles a dot: 4.43361875 MHz against a 128K's 7.0938 MHz and a 48K's 7
	float cycles() { return (CARRIER < 0.5) ? 0.625 : 0.63337411; }

	// the carrier's turn, radians, at a texel column of a line of the whole raster
	float turnAt(float tx, float line) {
		float dot_ = xpRaster.z + tx * 0.5;
		if (CARRIER < 0.5) return PI2 * (fract(0.625 * dot_) + PHASE * 0.125);
		// 0.4992 cycle further each frame: the two crystals do not keep step
		float frame = fract(float(xpFrame % 1250) * 0.4992);
		return PI2 * (fract(0.63337411 * dot_) + fract(0.7516 * line) + frame);
	}

	// the V switch's sense on a line
	float sense(float line) {
		float turned = (CARRIER < 0.5) ? float(xpFrame & 1) : 0.0;
		return (mod(line + turned, 2.0) < 0.5) ? 1.0 : -1.0;
	}

	float rand(vec2 co) { return fract(sin(dot(co, vec2(12.9898, 78.233))) * 43758.5453); }

	// SECAM. A line carries one of the two color differences, turn about, as the
	// frequency of a carrier: D'R = -1.902 (R - Y) about 4.40625 MHz, 280 kHz a unit,
	// D'B = 1.505 (B - Y) about 4.250 MHz, 230 kHz a unit, kept within +350 / -506 and
	// +506 / -350 kHz; the carrier is 0.115 of the luma's range at 4.286 MHz and larger
	// away from it (the "bell"). The set takes the band about 4.286 MHz, cares for the
	// carrier's frequency alone, and has the other difference from the line before (ITU-R
	// BT.470, table 2). A frequency has no phase to cancel by, and a set told nothing
	// apart from a carrier: fine luma - a dot on, a dot off is 3.5 MHz - near that band
	// pulls the reading its way, the same way on every line and every frame. That is the
	// flat color a pattern took on such a set. Cycles a texel, at 14 MHz:
	const float S_FC = 0.306143;
	const float S_FR = 0.314732;
	const float S_FB = 0.303571;
	const float S_DR = 0.020000;
	const float S_DB = 0.016429;
	const int S_TAPS = 12;

	// the carrier's frequency for a texel of the coder's picture on a line of one kind
	float secamSent(vec3 c, bool red) {
		vec2 uv = 2.0 * (c.gb - 128.0 / 255.0);
		if (red) return S_FR + S_DR * clamp(-1.902 * uv.y / 0.877, -1.807, 1.25);
		return S_FB + S_DB * clamp(1.505 * uv.x / 0.492, -1.522, 2.2);
	}

	float bell(float f) {
		float F = f / S_FC - S_FC / f;
		return sqrt((1.0 + 256.0 * F * F) / (1.0 + 1.5876 * F * F));
	}

	// the set's band about the carrier, as a window over the texels: a bell 4.5 texels
	// wide, brought down to nothing at the taps' ends
	float band(float d) {
		return exp(-0.5 * d * d / 20.25) * (0.5 + 0.5 * cos(3.14159265 * clamp(d / float(S_TAPS), -1.0, 1.0)));
	}

	// What the set reads on a line at a texel: the color difference of the line's kind
	// (.x), and the luma less the carrier's band (.y). The carrier is made as the walk
	// goes out from the texel both ways, its phase kept as it grows, so that inside the
	// set's window it is one whole carrier.
	vec2 secamRead(ivec2 at, bool red, float start) {
		int last = int(SourceSize.x) - 1;
		vec2 za = vec2(0.0), zb = vec2(0.0);
		float y = 0.0, trap = 0.0, wySum = 0.0, wnSum = 0.0;
		float sy = 1.2 - 0.8 * SHARP;
		float phase = start;
		float before = 0.0;
		for (int n = 0; n <= 2 * S_TAPS; n++) {
			// 0, 1 .. S_TAPS, then -1 .. -S_TAPS
			int k = (n <= S_TAPS) ? n : S_TAPS - n;
			vec3 c = texelFetch(Source, ivec2(clamp(at.x + k, 0, last), at.y), 0).rgb;
			float f = secamSent(c, red);
			if (n == 0) before = f;
			else if (n <= S_TAPS) phase += PI2 * f;
			else {
				if (n == S_TAPS + 1) phase = start;
				phase -= PI2 * before;
				before = f;
			}
			float s = c.r + 0.115 * bell(f) * cos(phase);
			float kf = float(k);
			// the set's band about 4.286 MHz, read half a texel left and half a texel
			// right of here: the turn between the two is the frequency
			// (the two windows are one window moved: anything else reads a frequency
			// pulled towards the band's middle)
			float edge = 0.5 + 0.5 * cos(3.14159265 * clamp(kf / (float(S_TAPS) + 0.5), -1.0, 1.0));
			// (the set's band filter - the bell turned over - leaves 0.17 of a 3.5 MHz
			// luma; this window leaves 0.47 of it: 0.36 of the luma goes in)
			vec2 mixed = (s - 0.64 * c.r) * vec2(cos(PI2 * S_FC * kf), -sin(PI2 * S_FC * kf));
			za += band(kf + 0.5) * mixed;
			zb += band(kf - 0.5) * mixed;
			float wy = exp(-0.5 * kf * kf / (sy * sy));
			float wn = exp(-0.5 * kf * kf / 9.0) * edge;
			y += wy * s;
			trap += wn * 2.0 * cos(PI2 * S_FC * kf) * s;
			wySum += wy;
			wnSum += wn;
		}
		float read = S_FC + atan(zb.y * za.x - zb.x * za.y, zb.x * za.x + zb.y * za.y) / PI2;
		// A discriminator is a tuned circuit: true inside the range the standard uses,
		// and past it the further off the frequency, the less it gives (as 1 / distance)
		float d = red ? (read - S_FR) / S_DR : (read - S_FB) / S_DB;
		float lo = red ? -1.807 : -1.522;
		float hi = red ? 1.25 : 2.2;
		if (d < lo) d = lo * lo / d;
		if (d > hi) d = hi * hi / d;
		return vec2(d, y / wySum - exp(-19.7392 * sy * sy * S_FC * S_FC) * trap / wnSum);
	}

	// the signal at a texel: luma with the color on its carrier
	float coded(int tx, int ty, float line, float sw) {
		vec3 c = texelFetch(Source, ivec2(clamp(tx, 0, int(SourceSize.x) - 1), ty), 0).rgb;
		float th = turnAt(float(tx), line);
		vec2 uv = 2.0 * (c.gb - 128.0 / 255.0);
		return c.r + uv.x * sin(th) + sw * uv.y * cos(th);
	}

	// ...as the cable brings it: with a ghost of itself, and an aerial's with noise
	float sent(int tx, int ty, float line, float sw) {
		float s = coded(tx, ty, line, sw);
		if (GHOST > 0.0) s += GHOST * 0.15 * coded(tx - int(GHOSTX * 0.01 * SourceSize.x), ty, line, sw);
		if ((NOISE > 0.0) && (INPUT > 1.5)) {
			float t = float(FrameCount % 1024u);
			s += NOISE * 0.08 * (rand(vec2(float(tx), float(ty)) + vec2(t * 17.13, t * 5.71)) - 0.5);
		}
		return s;
	}

	// ...and between two texels: the picture's own values mixed, the carrier where it is
	float sentAt(float fx, int ty, float line, float sw) {
		int last = int(SourceSize.x) - 1;
		float f = floor(fx);
		vec3 c = mix(texelFetch(Source, ivec2(clamp(int(f), 0, last), ty), 0).rgb,
			texelFetch(Source, ivec2(clamp(int(f) + 1, 0, last), ty), 0).rgb, fx - f);
		float th = turnAt(fx, line);
		vec2 uv = 2.0 * (c.gb - 128.0 / 255.0);
		float s = c.r + uv.x * sin(th) + sw * uv.y * cos(th);
		if (GHOST > 0.0) s += GHOST * 0.15 * coded(int(f) - int(GHOSTX * 0.01 * SourceSize.x), ty, line, sw);
		if ((NOISE > 0.0) && (INPUT > 1.5)) {
			float t = float(FrameCount % 1024u);
			s += NOISE * 0.08 * (rand(vec2(f, float(ty)) + vec2(t * 17.13, t * 5.71)) - 0.5);
		}
		return s;
	}

	void main() {
		ivec2 at = ivec2(vTexCoord * SourceSize.xy);
		int up = max(at.y - 1, 0);
		float line = xpRaster.w + float(at.y);
		if (INPUT > 2.5) {
			bool red = (mod(line, 2.0) < 0.5);
			// the carrier starts a line where it left the last, more or less: anywhere
			float start = PI2 * rand(vec2(line, float(xpFrame & 3)));
			vec2 mine = secamRead(at, red, start);
			vec2 above = secamRead(ivec2(at.x, up), !red, start + 2.0);
			float dR = red ? mine.x : above.x;
			float dB = red ? above.x : mine.x;
			float r = mine.y - COLOR * dR / 1.902;
			float b = mine.y + COLOR * dB / 1.505;
			float g = (mine.y - 0.299 * r - 0.114 * b) / 0.587;
			FragColor = vec4(clamp(vec3(r, g, b), 0.0, 1.0), float(xpFrame % 250) / 255.0);
			return;
		}
		float sw = sense(line);
		float swUp = sense(line - 1.0);
		bool rf = (INPUT > 1.5);
		// how wide the set's filters are, texels: the color's, the luma's, the trap's
		// (a set's video path is soft: an aerial input softer than a video one)
		float sc = rf ? 3.4 : 3.0;
		float sy = 1.9 - 1.2 * SHARP + (rf ? 0.8 : 0.0);
		const float sn = 3.0;
		const float late = 1.0;		// the color arrives a texel after its luma
		// The set's delay line holds the signal for 283.5 cycles of the carrier, 63.943
		// microseconds - a line of a broadcast, less a bit. A Spectrum's line is not a
		// broadcast's: a 128K's is 285 cycles long, so what comes out of the delay line
		// is the line above 4.8 texels to the right of here (a 48K's, 283.75 cycles:
		// 0.8). It is taken there, and by the carrier as it stands now - which is how
		// two lines of a pattern that an "average with the line above" would cancel
		// give the colors they gave.
		float delay = (CARRIER < 0.5) ? 4.8 : 0.79;
		float u = 0.0, v = 0.0, uUp = 0.0, vUp = 0.0, y = 0.0, trap = 0.0;
		float wcSum = 0.0, wySum = 0.0, wnSum = 0.0;
		for (int k = -TAPS; k <= TAPS; k++) {
			float kf = float(k);
			// the color's filter stands `late` to the left: the same texels serve both
			float wc = exp(-0.5 * (kf + late) * (kf + late) / (sc * sc));
			float wy = exp(-0.5 * kf * kf / (sy * sy));
			float wn = exp(-0.5 * kf * kf / (sn * sn));
			float s = sent(at.x + k, at.y, line, sw);
			y += wy * s;
			trap += wn * 2.0 * cos(PI2 * cycles() * 0.5 * kf) * s;
			float th = turnAt(float(at.x + k), line);
			u += wc * s * 2.0 * sin(th);
			v += wc * s * 2.0 * cos(th);
			float su = sentAt(float(at.x + k) + delay, up, line - 1.0, swUp);
			uUp += wc * su * 2.0 * sin(th);
			vUp += wc * su * 2.0 * cos(th);
			wcSum += wc;
			wySum += wy;
			wnSum += wn;
		}
		// the trap takes out what the low-pass has left of the carrier, no more
		float fc = cycles() * 0.5;
		y = y / wySum - exp(-19.7392 * sy * sy * fc * fc) * trap / wnSum;
		// the delayed line's carrier is half a cycle on, and its V the other way round:
		// the difference of the two is U, their sum V
		u = COLOR * 0.5 * (u - uUp) / wcSum;
		v = COLOR * 0.5 * sw * (v + vUp) / wcSum;
		float r = y + v / 0.877;
		float b = y + u / 0.492;
		float g = (y - 0.299 * r - 0.114 * b) / 0.587;
		// alpha: which frame this is, for the pass that mixes frames
		FragColor = vec4(clamp(vec3(r, g, b), 0.0, 1.0), float(xpFrame % 250) / 255.0);
	}

// The decoded frame with the one before, in linear light. The carrier's pattern turns
// over from one frame to the next - the V switch of a 128K, half a cycle of a 48K - and
// on a set the eye takes the two as one; on a screen that shows each frame whole, and
// not at the machine's own rate, they beat. So where the picture itself stands still
// the two frames are averaged, but for the part Shimmer leaves. Where it moves there is
// nothing to average, and antiflicker, left to the look while the frames come unmixed,
// has its own say: all of the picture, or - an adaptive mode - only where a texel
// is what it was two frames ago and not what it was one frame ago, which is a flicker
// and not a thing that moves.
// The same frame painted twice (the machine paused, a window redrawn) has no frame
// before it here: it is shown as it was shown.
pass seen:
	when INPUT
	filter nearest
fragment:
	in vec2 vTexCoord;
	layout(location = 0) out vec4 FragColor;

	void main() {
		vec4 before = texture(signalFeedback, vTexCoord);
		if (abs(before.a - texture(Source, vTexCoord).a) < 0.002) {
			FragColor = texture(seenFeedback, vTexCoord);
			return;
		}
		vec3 now = pow(texture(Source, vTexCoord).rgb, vec3(2.2));
		vec3 was = pow(before.rgb, vec3(2.2));
		// the frame's own texels here, as far as the decoder's filters reach that
		// matters, and on the line above: against a frame ago, and against two
		ivec2 at = ivec2(vTexCoord * OriginalSize.xy);
		ivec2 last = ivec2(OriginalSize.xy) - 1;
		float moved = 0.0;
		float back = 0.0;
		for (int k = -4; k <= 4; k++) {
			ivec2 p = clamp(at + ivec2(k, 0), ivec2(0), last);
			ivec2 q = clamp(at + ivec2(k, -1), ivec2(0), last);
			vec3 a = texelFetch(Original, p, 0).rgb;
			vec3 b = texelFetch(Original, q, 0).rgb;
			vec3 d = abs(a - texelFetch(OriginalHistory1, p, 0).rgb) + abs(b - texelFetch(OriginalHistory1, q, 0).rgb);
			vec3 e = abs(a - texelFetch(OriginalHistory2, p, 0).rgb) + abs(b - texelFetch(OriginalHistory2, q, 0).rgb);
			moved += d.r + d.g + d.b;
			back += e.r + e.g + e.b;
		}
		bool still = (moved < 0.004);
		float even = still ? 0.5 * (1.0 - SHIMMER) : 0.0;
		bool flicker = !still && (back < 0.004);
		float mixed = ((xpBlend.y < 0.5) || flicker) ? xpBlend.x : 0.0;
		FragColor = vec4(pow(mix(now, was, max(mixed, even)), vec3(1.0 / 2.2)), 1.0);
	}

pass afterglow:
	when AFTER
fragment:
	in vec2 vTexCoord;
	layout(location = 0) out vec4 FragColor;

	void main() {
		vec3 now = texture(Source, vTexCoord).rgb;
		vec3 was = texture(afterglowFeedback, vTexCoord).rgb;
		// fades in linear light, is kept encoded; a step of 8 bits less each frame, or the
		// last of it would stay for good
		vec3 left = was * pow(AFTER * vec3(AFTERR, AFTERG, AFTERB), vec3(1.0 / 2.2)) - 1.0 / 255.0;
		FragColor = vec4(max(now, left), 1.0);
	}

pass line:
	scale_x viewport 1.0
	scale_y source 1.0
	format rgba16f
	filter nearest
fragment:
	in vec2 vTexCoord;
	layout(location = 0) out vec4 FragColor;

	const float PI = 3.14159265;

	// A dot spreads along the line as a raised cosine of radius r texels. This is its
	// integral taken twice: what an edge at 0, lit to its right, gives left of t, summed
	// again over a stretch that ends at t. r = 0 is a hard edge.
	vec3 spread(vec3 t, float r) {
		vec3 c = clamp(t, -r, r);
		vec3 v = 0.5 * c + c * c / (4.0 * r) - r * cos(PI * c / r) / (2.0 * PI * PI) + 0.25 * r - r / (2.0 * PI * PI);
		return v + max(t - r, 0.0);
	}

	// The light of the screen pixel that stands at x texels along this line; each color's
	// beam lands at texels off its place, a dot spreads r texels. What spreads is the
	// drive, not the light: a video path too slow for a dot rounds the voltage, and the
	// tube makes light of what is left - so a lone lit dot comes out narrower than its
	// place and dimmer, a lone dark one stays dark, as on a tube; spread as light, lit
	// dots would swell into bricks and close the dark between them. One walk along the
	// texels for the three: a texel is read once, each color takes its own share of it
	vec3 lit(float x, vec3 at, float r) {
		float foot = SourceSize.x * OutputSize.z;		// a screen pixel, in texels
		vec3 from = x + at - 0.5 * foot;
		vec3 to = from + foot;
		int row = int(vTexCoord.y * SourceSize.y);
		int first = int(floor(min(from.r, min(from.g, from.b)) - r));
		int last = int(SourceSize.x) - 1;
		float end = max(to.r, max(to.g, to.b)) + r;
		// each texel's share of the pixel: what its left edge lights less what its right does
		vec3 left = spread(to - float(first), r) - spread(from - float(first), r);
		vec3 sum = vec3(0.0);
		vec3 all = vec3(0.0);
		for (int n = 0; n < 28; n++) {
			// past the pixel and the dot's spread nothing is left
			if (float(first + n) >= end) break;
			vec3 right = spread(to - float(first + n + 1), r) - spread(from - float(first + n + 1), r);
			vec3 w = max((left - right) / foot - 0.0005, vec3(0.0));
			left = right;
			vec3 c = texelFetch(Source, ivec2(clamp(first + n, 0, last), row), 0).rgb;
			sum += c * w;
			all += w;
		}
		return pow(sum / max(all, vec3(0.0001)), vec3(2.2));
	}

	void main() {
		float x = vTexCoord.x * SourceSize.x;
		// interference: a wave that runs down the screen pushes the lines aside
		if (WIGGLE > 0.0) {
			float y = vTexCoord.y;
			x += WIGGLE * 0.0012 * SourceSize.x * sin(turn(FrameCount, 573u) + y * 13.0)
				* sin(turn(FrameCount, 318u) + y * 19.0) * sin(0.3 + turn(FrameCount, 630u) + y * 23.0);
		}
		float r = max((1.0 - FOCUS) * 3.0, 0.0001);
		vec3 at = vec3(0.0);
		if (DIVERGE > 0.0) {
			// two texels a dot; no further than the walk along the line reaches
			at.r = clamp(2.0 * parting(vTexCoord, vec2(DIVRX, DIVRY)).x, -8.0, 8.0);
			at.b = clamp(2.0 * parting(vTexCoord, vec2(DIVBX, DIVBY)).x, -8.0, 8.0);
		}
		vec3 col = lit(x, at, r);
		// the ghost: the picture once more to the right, faint and out of focus - a glow
		// that stands aside, not a second picture
		// (a PAL picture has its ghost in the signal)
		if ((GHOST > 0.0) && ((INPUT == 0.0) || (INPUT > 2.5))) col += GHOST * 0.15 * lit(x - GHOSTX * 0.01 * SourceSize.x, vec3(0.0), 4.0);
		FragColor = vec4(pow(min(col, vec3(1.0)), vec3(1.0 / 2.2)), 1.0);
	}

pass glowh:
	scale viewport 0.25
	filter linear
	wrap clamp
	when GLOW
fragment:
	in vec2 vTexCoord;
	layout(location = 0) out vec4 FragColor;

	void main() {
		vec3 sum = vec3(0.0);
		float all = 0.0;
		for (int n = -4; n <= 4; n++) {
			float w = exp(-0.125 * float(n * n));
			sum += pow(texture(Source, vTexCoord + vec2(float(n) * GLOWR * OutputSize.z, 0.0)).rgb, vec3(2.2)) * w;
			all += w;
		}
		FragColor = vec4(pow(sum / all, vec3(1.0 / 2.2)), 1.0);
	}

pass glowv:
	scale viewport 0.25
	filter linear
	wrap clamp
	when GLOW
fragment:
	in vec2 vTexCoord;
	layout(location = 0) out vec4 FragColor;

	void main() {
		vec3 sum = vec3(0.0);
		float all = 0.0;
		for (int n = -4; n <= 4; n++) {
			float w = exp(-0.125 * float(n * n));
			sum += pow(texture(Source, vTexCoord + vec2(0.0, float(n) * GLOWR * OutputSize.w)).rgb, vec3(2.2)) * w;
			all += w;
		}
		FragColor = vec4(pow(sum / all, vec3(1.0 / 2.2)), 1.0);
	}

pass:
	filter linear
fragment:
	in vec2 vTexCoord;
	layout(location = 0) out vec4 FragColor;

	const float PI2 = 6.28318530718;

	// a line of the line pass at x, in linear light; past the first and the last line,
	// that line: a beam that lands a line off keeps its color at the picture's edge
	vec3 lineAt(float x, float row) {
		row = clamp(row, 0.0, lineSize.y - 1.0);
		return pow(texture(line, vec2(x, (row + 0.5) * lineSize.w)).rgb, vec3(2.2));
	}

	// the picture at a point: the two lines the screen pixel stands between, by the share
	// each has of it. foot: the pixel's height in lines
	vec3 picture(vec2 uv, float foot) {
		float b = uv.y * lineSize.y - 0.5;
		float i = floor(b);
		float k = snap(clamp((b - i - 0.5) / min(foot, 1.0) + 0.5, 0.0, 1.0));
		return mix(lineAt(uv.x, i), lineAt(uv.x, i + 1.0), k);
	}

	// A scanline's gap is a dip of the shape of a cosine, as deep as depth and as wide
	// as width of the line's height, half of it at each end of the line. This is the
	// light from the line's top down to x of its height, inside the dip (x <= width / 2)
	vec3 dip(vec3 x, vec3 depth, float width) {
		return x - 0.5 * depth * (x + width / PI2 * sin(PI2 * x / width));
	}

	// ...and down to u anywhere, for a line of the strength amp: from tc to 1 - tc it
	// is flat - out of the dip, or where it would pass full light
	// atTc: dip() at tc
	vec3 light(vec3 amp, vec3 tc, vec3 atTc, float u, vec3 depth, float width) {
		vec3 uu = vec3(u);
		vec3 head = amp * dip(min(uu, tc), depth, width);
		vec3 flat_ = min(amp, vec3(1.0)) * clamp(uu - tc, vec3(0.0), 1.0 - 2.0 * tc);
		vec3 tail = amp * (atTc - dip(clamp(1.0 - uu, vec3(0.0), tc), depth, width));
		return head + flat_ + tail;
	}

	vec2 warp(vec2 uv, float bend) {
		vec2 p = uv * 2.0 - 1.0;
		p *= vec2(1.0 + p.y * p.y * bend * CURVEX, 1.0 + p.x * p.x * bend * CURVEY);
		return p * 0.5 + 0.5;
	}

	float rand(vec2 co) {
		return fract(sin(dot(co, vec2(12.9898, 78.233))) * 43758.5453);
	}

	// A triad is as wide as the tube's fineness makes it, in whole stripes of screen
	// pixels: red, green and blue, each n wide. With under two and a half pixels to a
	// triad there is no room for three: two colors a pair, magenta and green.
	// scale: screen pixels a dot. Returns the triad's width
	float triad(float scale, out bool pair, out float n) {
		float room = scale / PITCH;
		pair = (room < 2.5);
		n = max(1.0, floor(room / 3.0 + 0.5));
		return pair ? 2.0 : 3.0 * n;
	}

	// What the mask lets through at a screen pixel, for each color, and how much of the
	// light it lets through over all (what is given back).
	// lines: screen rows a scanline takes, a whole number. Where the scanlines follow
	// the picture (a curved screen, Line fit) they take a little more or less, and a
	// slot mask still beats with them, faintly
	vec3 mask(vec2 px, float cell, bool pair, float n, float lines, out float mean) {
		mean = 1.0;
		float dark = 1.0 - MASKS;
		float x = px.x;
		float gap = 1.0;				// the slot's bar between two openings
		if (MASK == 2.0) {
			// openings about four stripes tall, a bar a stripe tall between them; every
			// other triad half an opening lower. As tall as a whole number of scanlines:
			// any other height beats with them
			float want = pair ? 4.0 : 4.0 * n;
			float tall = lines * max(1.0, floor(want / lines + 0.5));
			float bar = pair ? 1.0 : n;
			float y = px.y + mod(floor(x / cell), 2.0) * floor(0.5 * tall);
			if (mod(y, tall) < bar) gap = dark;
			mean *= 1.0 - MASKS * bar / tall;
		} else if (MASK == 3.0) {
			// rows of dots, every other row half a triad along
			float tall = pair ? 1.0 : 2.0 * n;
			x += mod(floor(px.y / tall), 2.0) * 0.5 * cell;
		}
		vec3 m = vec3(dark);
		if (pair) {
			if (mod(floor(x), 2.0) < 1.0) m.rb = vec2(1.0); else m.g = 1.0;
			mean *= 1.0 - 0.5 * MASKS;
		} else {
			float s = mod(floor(x / n), 3.0);
			if (s < 1.0) m.r = 1.0; else if (s < 2.0) m.g = 1.0; else m.b = 1.0;
			mean *= 1.0 - MASKS * 2.0 / 3.0;
		}
		return m * gap;
	}

	void main() {
		// the curve is kept to the picture: drawn on past its edges (Side fill: Stretched
		// Edges, Blur Off) the coordinates run on straight
		vec2 inPic = clamp(vTexCoord, vec2(0.0), vec2(1.0));
		// a small picture has no room for a curve: it comes in between two and a half
		// and three and a half pixels a dot
		float dot_ = 1.0 / (lineSize.y * OutputSize.w);
		float bend = CURVE * smoothstep(2.5, 3.5, dot_);
		bool flat_ = (bend == 0.0);
		vec2 uv = flat_ ? vTexCoord : warp(inPic, bend) + (vTexCoord - inPic);

		// a screen pixel's height in lines, and what the picture has there; the red and
		// blue beams from a little off
		float foot = flat_ ? lineSize.y * OutputSize.w : max(fwidth(uv.y) * lineSize.y, 0.0001);
		float scale = 1.0 / foot;
		vec3 col = picture(uv, foot);
		if (DIVERGE > 0.0) {
			col.r = picture(uv + vec2(0.0, parting(uv, vec2(DIVRX, DIVRY)).y * lineSize.w), foot).r;
			col.b = picture(uv + vec2(0.0, parting(uv, vec2(DIVBX, DIVBY)).y * lineSize.w), foot).b;
		}
		col *= BRIGHT;

		// The mask: counted in the window's pixels, or in the picture's own where it is
		// to bend with the glass - by itself where a triad has six pixels or more, under
		// which a bent mask beats with them
		bool pair;
		float n;
		float cell = triad(dot_, pair, n);
		bool glass = !flat_ && ((MBEND == 2.0) || ((MBEND == 0.0) && (cell >= 6.0)));
		vec2 px = glass ? uv * OutputSize.xy : gl_FragCoord.xy - xpPicture.xy;
		float mean;
		vec3 through = mask(px, cell, pair, n, max(2.0, floor(dot_ + 0.5)), mean);

		// A scanline is a line of the picture, at any scale: one that kept to whole
		// screen rows would cut across the picture's lines. On a flat screen at a whole
		// scale the two are the same thing and the rows are counted. Lines that do not
		// fall on rows beat with them at a small scale: shallower there
		bool rows = flat_ && (abs(scale - floor(scale + 0.5)) < 0.01);
		float deep = SCAN * (rows ? 1.0 : mix(0.4, 1.0, smoothstep(2.0, 4.0, scale)) * step(1.5, scale));
		vec3 depth = max(deep * (1.0 - GROW * col), vec3(0.0001));
		// what the gaps and the mask take is given back
		vec3 amp = max(col * through / mean / (1.0 - 0.5 * depth * WIDTH), vec3(0.000001));
		// where the dip ends, or sooner where the line would pass full light
		vec3 tc = WIDTH * acos(clamp((2.0 / depth) * (1.0 - 1.0 / amp) - 1.0, -1.0, 1.0)) / PI2;
		vec3 atTc = dip(tc, depth, WIDTH);
		vec3 whole = 2.0 * amp * atTc + min(amp, vec3(1.0)) * (1.0 - 2.0 * tc);
		// the pixel's share of the scanline, the gap on its last row; a roll moves the
		// lines down the screen
		float period = rows ? max(1.0, floor(scale + 0.5)) : scale;
		float row = rows ? vTexCoord.y * OutputSize.y : uv.y * lineSize.y * scale;
		// a step of Roll is a turn in 400 frames: the lines meet themselves when the
		// count of frames starts again
		if (ROLL > 0.0) row -= fract(float(FrameCount % 400u) * floor(ROLL * 20.0 + 0.5) / 400.0) * period;
		float from = row / period;
		float to = (row + 1.0) / period;
		vec3 sum = (floor(to) - floor(from)) * whole
			+ light(amp, tc, atTc, fract(to), depth, WIDTH) - light(amp, tc, atTc, fract(from), depth, WIDTH);
		vec3 lit = max(sum * period, vec3(0.0));

		// The light that spills around bright things: a part of each point's light is
		// spread about, so a plain field keeps its brightness and the dark beside a
		// bright thing gets a veil - at most a quarter
		if (GLOW > 0.0) {
			vec3 around = pow(texture(glowv, uv).rgb, vec3(2.2));
			lit = mix(lit, around, 0.25 * GLOW);
		}
		// (an aerial's noise is in the signal)
		if ((NOISE > 0.0) && ((INPUT < 1.5) || (INPUT > 2.5))) {
			// another grain each frame, not the same one moved along
			float t = float(FrameCount % 1024u);
			vec2 seed = gl_FragCoord.xy + vec2(t * 17.13, t * 5.71);
			lit = max(lit - 0.03 * NOISE * pow(vec3(rand(seed), rand(seed * 2.0), rand(seed * 3.0)), vec3(1.5)), vec3(0.0));
		}

		// the screen's own edge: past a bent picture's top and bottom nothing is lit
		// (its sides are the line pass's edge); corners rounded in the window's pixels,
		// a vignette
		if (xpWide < 0.5) {
			if (!flat_) lit *= clamp(min(uv.y, 1.0 - uv.y) * lineSize.y * scale + 0.5, 0.0, 1.0);
			vec2 size = OutputSize.xy;
			float rad = CORNER * min(size.x, size.y);
			vec2 q = abs(vTexCoord - 0.5) * size - (0.5 * size - rad);
			float out_ = length(max(q, vec2(0.0))) + min(max(q.x, q.y), 0.0) - rad;
			lit *= clamp(0.5 - out_, 0.0, 1.0);
			vec2 u = inPic;
			lit *= mix(1.0, pow(16.0 * u.x * u.y * (1.0 - u.x) * (1.0 - u.y), 0.25), VIGN);
		}
		vec3 seen = pow(lit, vec3(1.0 / 2.2));
		if (CONTRAST > 0.0) seen = mix(seen, seen * seen * (3.0 - 2.0 * seen), CONTRAST);
		FragColor = vec4(seen, 1.0);
	}
